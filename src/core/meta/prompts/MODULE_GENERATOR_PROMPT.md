                            "execution_time": time.time() - start_time

                "output": result_data,
                "error": None,

                    "module": self.module_name,
                    "execution_time": time.time() - start_time

        except SpecificException1 as e:

                "error": {
                    "type": "SpecificException1",

                    "details": {}

## 🚫 STRICT RULES (嚴格規則)

### ❌ RULE 1: NO Duplicate Imports

**English:**
- ALL imports MUST be at file top
- NEVER import inside `execute()`
- NEVER import inside `validate_params()`

**中文：**
- 所有 import 必須在文件頂部
- 絕不在 `execute()` 內 import
- 絕不在 `validate_params()` 內 import

```python
# ❌ WRONG
async def execute(self):
    import httpx  # ← NEVER do this!
    from pathlib import Path  # ← NEVER do this!

# ✅ CORRECT
import httpx
from pathlib import Path

async def execute(self):
    # No imports here
```

**Deduction:** -1.0 point

---

### ❌ RULE 2: ALWAYS Use self.{variable}

**English:**
- ALL class properties MUST use `self.` prefix
- NEVER use bare variable names

**中文：**
- 所有類屬性必須使用 `self.` 前綴
- 絕不使用裸變數名

```python
# ❌ WRONG
if not url.startswith("http://"):  # ← Missing self.!
    path = Path(save_path)  # ← Missing self.!

# ✅ CORRECT
if not self.url.startswith("http://"):
    path = Path(self.save_path)
```

**Deduction:** -0.2 per instance (max -1.0)

---

### ❌ RULE 3: NO Nested Functions

**English:**
- NEVER define functions inside `execute()`
- NEVER define functions inside `validate_params()`
- Use class methods if needed

**中文：**
- 絕不在 `execute()` 內定義函數
- 絕不在 `validate_params()` 內定義函數
- 如需要，使用類方法

```python
# ❌ WRONG
async def execute(self):
    def helper():  # ← NEVER do this!
        pass

# ✅ CORRECT
def _helper_method(self, data):
    # Define as class method
    pass

async def execute(self):
    result = self._helper_method(data)
```

**Deduction:** -0.5 point

---

### ❌ RULE 4: validate_params() = VALIDATION ONLY

**English:**
- ONLY check parameter existence
- ONLY assign to self.{param}
- NO business logic
- NO API calls
- NO file operations

**中文：**
- 僅檢查參數是否存在
- 僅賦值給 self.{param}
- 無業務邏輯
- 無 API 調用
- 無文件操作

```python
# ❌ WRONG
def validate_params(self):
    self.url = self.params["url"]
    response = httpx.get(self.url)  # ← NO business logic!

# ✅ CORRECT
def validate_params(self):
    if "url" not in self.params:
        raise ValueError("Missing required parameter: url")
    self.url = self.params["url"]
    # Only validation, no logic
```

**Deduction:** -1.0 point

**中文：**

### ❌ RULE 6: Use Async I/O for Network

**English:**
- Network operations MUST use async libraries (httpx, aiohttp)
- NEVER use synchronous libraries (requests, urllib)

**中文：**
- 網絡操作必須使用異步庫（httpx, aiohttp）
- 絕不使用同步庫（requests, urllib）

```python
# ❌ WRONG
import requests
response = requests.get(url)  # ← Blocking!

# ✅ CORRECT
import httpx
async with httpx.AsyncClient() as client:
    response = await client.get(self.url, timeout=30.0)
```

**Deduction:** -1.0 point

### ❌ RULE 7: Catch Specific Exceptions

**English:**
- Catch at least 3 specific exception types
- Avoid generic `except Exception` as the only handler

**中文：**
- 捕獲至少 3 種具體異常類型
- 避免僅使用通用 `except Exception`

```python
# ❌ WRONG
try:
    result = await client.get(url)
except:  # ← Too broad!
    pass

# ✅ CORRECT
try:
    result = await client.get(self.url)
except httpx.HTTPStatusError as e:
    return {"ok": False, "error": {"type": "HTTPStatusError", "message": str(e)}, ...}
except httpx.RequestError as e:
    return {"ok": False, "error": {"type": "RequestError", "message": str(e)}, ...}
except IOError as e:
    return {"ok": False, "error": {"type": "IOError", "message": str(e)}, ...}
except Exception as e:
    return {"ok": False, "error": {"type": "UnexpectedError", "message": str(e)}, ...}
```

**Deduction:** -1.0 point

### ❌ RULE 8: Security Validations

**For URL parameters:**
- MUST validate URL format (http/https)
- MUST validate Content-Type (for downloads)
- MUST check file size limits (for downloads)

**For file parameters:**
- MUST prevent path traversal
- MUST validate file extensions
- MUST enforce size limits

**中文：**

**URL 參數：**
- 必須驗證 URL 格式（http/https）
- 必須驗證 Content-Type（下載時）
- 必須檢查文件大小限制（下載時）

**文件參數：**
- 必須防止路徑穿越
- 必須驗證文件擴展名
- 必須強制執行大小限制

```python
# ✅ CORRECT (URL validation)
if not (self.url.startswith("http://") or self.url.startswith("https://")):
    return {
        "ok": False,
        "error": {"message": "Invalid URL format"},
        ...
    }

# ✅ CORRECT (Content-Type check for downloads)
head_response = await client.head(self.url)
content_type = head_response.headers.get("content-type", "").lower()
if not content_type.startswith("image/"):
    return {
        "ok": False,
        "error": {"message": f"Invalid content type: {content_type}"},
        ...
    }

# ✅ CORRECT (File size limit)
content_length = int(head_response.headers.get("content-length", 0))
max_size = 50 * 1024 * 1024  # 50 MB
if content_length > max_size:
    return {
        "ok": False,
        "error": {"message": f"File too large: {content_length} bytes"},
        ...
    }
```

**Deduction:** -0.5 per missing validation
